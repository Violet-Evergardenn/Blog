### Exploring Code Snippets

Here is how you might write a simple fetch in React:

```tsx
import { useEffect, useState } from 'react';

export function useFetchPost(id: string) {
  const [content, setContent] = useState('');
  
  useEffect(() => {
    fetch(`/posts/${id}.md`)
      .then(res => res.text())
      .then(text => setContent(text));
  }, [id]);

  return content;
}
```
